从 $m$ 到 $1$ 倒序合并，如果 $a_i$ 和 $b_i$ 得关系已经确定就跳过。

否则，考虑是否放到同一个班，需要满足最终两班人数恰好为 $n$ 的限制。

思考一下，发现判断形如有多个二元组 $(x_i, y_i)$，我们要从每个二元组中选一个数使得和为 $n$。先让 $n$ 减去 $\sum \min(x_i, y_i)$，然后相当于体积为 $|x_i - y_i|$ 的物品做 01 背包，可以 bitset 做到 $O(\frac {n^2\sqrt n}w)$，但是更简单的做法是直接退背包，考虑 $f_i$ 表示凑出 $i$ 的方案数，每次判断会修改 $O(1)$ 个二元组，直接从背包里删掉，方案数不为 $0$ 即可，这样是 $O(n^2)$ 的。


```cpp
#include <bits/stdc++.h>
using namespace std;
constexpr int N = 5e3 + 9, M = 1e6 + 9, md = 998244353;
int n, m, a[M], b[M], fa[N * 4], sz[N * 4], f[N], up;
int get(int x) { return x == fa[x] ? x : fa[x] = get(fa[x]); }
void add(int x, int y) {
  up -= min(x, y);
  int t = abs(x - y);
  if (t == 0) return ;
  for (int i = n / 2; i >= t; --i) f[i] += f[i - t], f[i] >= md && (f[i] -= md);
}
void del(int x, int y) {
  up += min(x, y);
  int t = abs(x - y);
  if (t == 0) return ;
  for (int i = t; i <= n / 2; ++i) f[i] -= f[i - t], f[i] < 0 && (f[i] += md);
}
void merge(int x, int y) {
  int p1 = get(x), p2 = get(y), p3 = get(x + n), p4 = get(y + n);
  if (p1 == p2 || p1 == p4) return ;
  add(sz[p1] + sz[p4], sz[p2] + sz[p3]);
  del(sz[p1], sz[p3]), del(sz[p2], sz[p4]);
  if (f[up]) {
    if (sz[p2] > sz[p3]) swap(p2, p3);
    if (sz[p1] > sz[p4]) swap(p1, p4);
    fa[p1] = p4, fa[p2] = p3, sz[p4] += sz[p1], sz[p3] += sz[p2];
  }
  else {
    del(sz[p1] + sz[p4], sz[p2] + sz[p3]);
    add(sz[p1] + sz[p2], sz[p3] + sz[p4]);
    if (sz[p1] > sz[p2]) swap(p1, p2);
    if (sz[p3] > sz[p4]) swap(p3, p4);
    fa[p1] = p2, fa[p3] = p4, sz[p4] += sz[p3], sz[p2] += sz[p1];
  }
}
signed main() {
  cin.tie(nullptr)->sync_with_stdio(false);
  cin >> n >> m, n <<= 1;
  f[0] = 1, up = n / 2;
  for (int i = 1; i <= m; ++i) cin >> a[i] >> b[i];
  for (int i = 1; i <= 2 * n; ++i) fa[i] = i, sz[i] = i <= n;
  for (int i = 1; i <= n; ++i) add(0, 1);
  for (int i = m; i >= 1; --i) merge(a[i], b[i]);
  for (int i = 2; i <= n; ++i) merge(i, 1);
  for (int i = 1; i <= n; ++i) cout << (get(i) == get(1));
}
```
